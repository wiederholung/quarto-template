% Local, overridable author-block defaults. The class owns the title and date.
\providecommand{\AuthorsBlockNoteFont}{\small}
\providecommand{\AuthorsBlockIconScale}{1}
\providecommand{\AuthorsBlockIconGap}{.15em}
\providecommand{\AuthorsBlockMarkGap}{.08em}
\providecommand{\AuthorsBlockGroupGap}{.45\baselineskip}
\providecommand{\AuthorsBlockNoteGap}{.15\baselineskip}
\newlength{\AuthorsBlockCapHeight}
\newsavebox{\AuthorsBlockPersonBox}
\providecommand{\AuthorsBlockPerson}[1]{\begingroup
  \sbox{\AuthorsBlockPersonBox}{#1}%
  \ifdim\wd\AuthorsBlockPersonBox>\linewidth #1\else\usebox{\AuthorsBlockPersonBox}\fi
  \endgroup}
\providecommand{\AuthorsBlockIcon}[1]{%
  \begingroup\settoheight{\AuthorsBlockCapHeight}{H}%
  \includegraphics[height=\AuthorsBlockIconScale\AuthorsBlockCapHeight]{#1}\endgroup}
\providecommand{\AuthorsBlockMark}[1]{\nobreak\hspace{\AuthorsBlockMarkGap}\textsuperscript{#1}}
\providecommand{\AuthorsBlockNote}[2]{\par\vspace{\AuthorsBlockNoteGap}%
  {\AuthorsBlockNoteFont #1\nobreak\hspace{.25em}#2\par}}
\providecommand{\AuthorsBlockLayout}[2]{%
  \begin{minipage}[t]{\dimexpr\textwidth-2\tabcolsep\relax}%
  \centering #1\par
  \if\relax\detokenize{#2}\relax\else\vspace{\AuthorsBlockGroupGap}#2\fi
  \end{minipage}}
